\begin{Purpose}
To understand the fundamentals of financial mathematics

Preparation for SOA/CAS exam FM/2 and other professional certification exams.
\end{Purpose}

\begin{Text}
\emph{Mathematics of Investment \& Credit}, 7\textsuperscript{th} edition, by Broverman, published by Actex. ISBN-13: 9781635882216
\end{Text}

\begin{TopicsList}
\item Compound and simple rates of interest and discount, force of interest.
\item Annuities.
\item Mortgage amortization.
\item Valuation of bonds.
\item Duration.
\item Asset and liability matching.
\item Cash flows.
\end{TopicsList}
