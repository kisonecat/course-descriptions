\begin{Purpose}
This course provides a comprehensive introduction to complex analysis,
emphasizing applications that are useful in science and engineering.
\end{Purpose}

\begin{Text}
\emph{Complex Variables and Applications}, 8\textsuperscript{th}
edition, by Brown \& Churchill, published by McGraw-Hill, ISBN:
9780073383170
\end{Text}

\begin{TopicsOutline}
Complex numbers, polar form (Ch. 1)

Analyticity, Cauchy-Riemann equations (Ch. 2)

Elementary functions (Ch. 3)

Cauchy integral theorem and consequences (Ch. 4)

\emph{Midterm 1}

Power series (Ch. 5)

Residues and poles (Ch. 6)

Applications of residues (Ch. 7)

Mapping by elementary functions (Ch. 8)

Conformal mappings (Ch. 9)

\emph{Midterm 2}

Applications of conformal mappings (Ch. 10)

Schwarz-Christoffel transformation (Ch. 11)

Poisson integral, Dirichlet problem (Ch. 12)
\end{TopicsOutline}

\begin{SuggestedSchedule}
\week{1}{Ch. 1 Complex numbers; polar form; roots.}
\week{2}{Ch. 2 Analyticity; Cauchy-Riemann equations; harmonic functions.}
\week{3}{Ch. 3 Elementary functions: exponential, logarithm, trigonometric.}
\week{4}{Ch. 4 Contour integration; Cauchy-Goursat theorem.}
\week{5}{Ch. 4 Cauchy integral formula; Liouville's theorem. \emph{Midterm 1.}}
\week{6}{Ch. 5 Taylor and Laurent series.}
\week{7}{Ch. 6 Residues and poles.}
\week{8}{Ch. 7 Evaluation of improper and trigonometric integrals.}
\week{9}{Ch. 7 Indented paths; inverse Laplace transforms.}
\week{10}{Ch. 8 Mapping by elementary functions.}
\week{11}{Ch. 9 Conformal mappings. \emph{Midterm 2.}}
\week{12}{Ch. 10 Applications of conformal mappings.}
\week{13}{Ch. 11 Schwarz-Christoffel transformation.}
\week{14}{Ch. 12 Poisson integral; the Dirichlet problem.}
\finalsweek{Final examination.}
\end{SuggestedSchedule}
