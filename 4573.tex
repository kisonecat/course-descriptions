\begin{Purpose}
To introduce students to some topics in number theory at the upper
undergraduate level and make connections to other areas of mathematics,
such as combinatorics and abstract algebra.
\end{Purpose}

\begin{Text}
\emph{An Introduction to the Theory of Numbers}, 5\textsuperscript{th}
edition, by Niven, Zuckerman \& Montgomery, published by TBS, ISBN:
9780471625469
\end{Text}

\begin{TopicsList}
\item
  Prime numbers and factorization
\item
  Congruences and modular arithmetic; the Euler phi-function
  \(\varphi(n)\).
\item
  Fermat's ``Little'' Theorem. Primitive roots.
\item
  Quadratic Reciprocity
\item
  Numerical Functions of number theory; multiplicative functions and
  Moebius inversion
\item
  Diophantine equations.
\item
  Number theory from an algebraic viewpoint; groups, rings and fields.
\item
  Possible additional topics: continued fractions, Pell's equation, and
  elliptic curves.
\end{TopicsList}

\begin{SuggestedSchedule}
\week{1}{Divisibility, the Euclidean algorithm, unique factorization.}
\week{2}{Distribution of primes; primality and factoring.}
\week{3}{Congruences and modular arithmetic.}
\week{4}{Linear congruences; the Chinese Remainder Theorem.}
\week{5}{Euler's phi-function; Fermat's Little Theorem; Euler's theorem.}
\week{6}{Primitive roots and indices. \emph{Midterm I.}}
\week{7}{Quadratic residues; Euler's criterion.}
\week{8}{Quadratic reciprocity.}
\week{9}{Multiplicative functions; Moebius inversion.}
\week{10}{Sums of divisors; perfect numbers.}
\week{11}{Diophantine equations; Pythagorean triples. \emph{Midterm II.}}
\week{12}{Sums of squares; Fermat descent.}
\week{13}{Number theory from an algebraic viewpoint: groups, rings, fields.}
\week{14}{Additional topics: continued fractions, Pell's equation, elliptic curves.}
\finalsweek{Final examination.}
\end{SuggestedSchedule}
