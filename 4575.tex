\begin{Purpose}
Combinatorics and discrete mathematics are increasingly important,
particularly for their applications in computer science. This course
will give a brief overview of this subject.
\end{Purpose}

\begin{Text}
\emph{Introductory Combinatorics} 5\textsuperscript{th} Edition, by
Richard A. Brualdi, Pearson, ISBN: 9780136020400
\end{Text}

\begin{TopicsList}
\item
  Counting principles: factorials, permutations and combinations,
  binomial coefficients, Stirling numbers, double counting.
\item
  Combinatorial identities: bijections, binomial theorem, generating
  functions.
\item
  Graph theory: bridges of Konigsberg, Eulerian circuits, trees, edge
  coloring, vertex coloring, planar graphs, Kempe's proof of the 5-color
  theorem
\item
  Error correcting codes: sphere packing bound, Hamming codes
\item
  (Optional.) Algorithms: Djkstra's algorithm for minimum spanning tree,
  depth first and breadth first algorithms for trees, greedy algorithm
  for graph coloring.
\end{TopicsList}

\begin{SuggestedSchedule}
\week{1}{Counting principles; permutations and combinations.}
\week{2}{Binomial coefficients; Pascal's triangle; the binomial theorem.}
\week{3}{Inclusion-exclusion; double counting.}
\week{4}{Distributions, partitions, and Stirling numbers.}
\week{5}{Bijective proofs and combinatorial identities.}
\week{6}{Recurrence relations; generating functions. \emph{Midterm I.}}
\week{7}{Graphs: bridges of Konigsberg, Eulerian and Hamiltonian circuits.}
\week{8}{Trees and spanning trees.}
\week{9}{Matchings and edge coloring.}
\week{10}{Vertex coloring; chromatic polynomials.}
\week{11}{Planar graphs; Kempe's proof of the 5-color theorem. \emph{Midterm II.}}
\week{12}{Error correcting codes; the sphere packing bound.}
\week{13}{Hamming codes.}
\week{14}{Optional algorithms: minimum spanning trees, graph search, greedy coloring.}
\finalsweek{Final examination.}
\end{SuggestedSchedule}
