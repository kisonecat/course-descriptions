\begin{Purpose}
To provide students with a solid foundation in one-variable calculus, to introduce multi-variable tools in a biological setting, to model and analyze phenomena in the life sciences.
\end{Purpose}

\begin{Text}
\emph{Calculus for Biology and Medicine}, 3\textsuperscript{rd} Edition, by Claudia Neuhauser, Pearson, ISBN 9780321644688
\end{Text}
