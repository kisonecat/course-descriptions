\begin{Text}
\emph{Journey through Genius}, by William Dunham, published by Wiley,
ISBN: 0471500305, and

\emph{Ximera History of Mathematics} (Free online access through Carmen)
\end{Text}

\begin{TopicsList}
\item
  Development of arithmetic; Babylonian tablets and Egyptian papyri.
\item
  Development of geometry: Pythagoras, Thales, Euclid, Archimedes, and
  non-Euclidean geometry.
\item
  Development of algebra: Arabic scholars, Fibonacci, Cardano.
\item
  Development of calculus: Newton, Leibniz, Bernoulli, Euler, Gauss.
\item
  Beyond calculus: Cantor, others by choice of instructor or students.
\end{TopicsList}

\begin{SuggestedSchedule}
\week{1}{Origins of counting; Babylonian tablets.}
\week{2}{Egyptian papyri; development of arithmetic.}
\week{3}{Thales and Pythagoras.}
\week{4}{Euclid's \emph{Elements}.}
\week{5}{Archimedes.}
\week{6}{The parallel postulate and non-Euclidean geometry.}
\week{7}{Midterm; Arabic scholars and the rise of algebra.}
\week{8}{Fibonacci; Cardano and the cubic.}
\week{9}{Descartes, Fermat, and analytic geometry.}
\week{10}{Newton.}
\week{11}{Leibniz and the Bernoullis.}
\week{12}{Euler.}
\week{13}{Gauss.}
\week{14}{Cantor and the infinite; additional topics by choice.}
\finalsweek{Final examination or project presentations.}
\end{SuggestedSchedule}
