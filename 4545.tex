\begin{Text}
\emph{Introduction to Real Analysis}, by William F. Trench, Edition
1.03, published by Library of Congress Cataloging-in-Publication Data,
ISBN: 0130457868
\end{Text}

\begin{TopicsList}
\item
  Limits and continuity of functions.
\item
  Derivative, mean value theorem, optimization.
\item
  Sequences and series, uniform convergence, power series, Taylor's
  theorem.
\item
  Riemann integral, substitution, bounded variation, limit properties,
  Rieman-Stieltjes integral.
\item
  Multivariable functions, directional derivatives, chain rule, Taylor's
  theorem.
\item
  Inverse and implicit function theorems.
\end{TopicsList}

\begin{SuggestedSchedule}
\week{1}{The real numbers; completeness; limits of sequences.}
\week{2}{Limits and continuity of functions.}
\week{3}{Continuity on compact sets; uniform continuity.}
\week{4}{The derivative; mean value theorem.}
\week{5}{Optimization; Taylor's theorem in one variable.}
\week{6}{Series; tests for convergence. \emph{Midterm I.}}
\week{7}{Sequences and series of functions; uniform convergence.}
\week{8}{Power series.}
\week{9}{The Riemann integral; substitution.}
\week{10}{Bounded variation; limit properties of the integral.}
\week{11}{The Riemann-Stieltjes integral. \emph{Midterm II.}}
\week{12}{Functions of several variables; directional derivatives.}
\week{13}{Chain rule; Taylor's theorem in several variables.}
\week{14}{Inverse and implicit function theorems.}
\finalsweek{Final examination.}
\end{SuggestedSchedule}
