\begin{Purpose}
This course provides students with experience and knowledge in mathematical analysis of differential equations models, as well as with numerical tools for simulating those models.
\end{Purpose}

\begin{Text}
\emph{Lecture Notes} and

\emph{Introduction to Mathematical Biology (Modeling, Analysis, and Simulations)}, by Ching-Shan Chou and Avner Friedman, Springer International Publishing, 2016. (eBook ISBN: 9783319296388; Hardcover ISBN: 9783319296364)
\end{Text}

\begin{TopicsList}
\item Population dynamics: Logistic growth.
\item Population dynamics: Lotka-Volterra predator-prey model.
\item Modeling infectious diseases.
\item Competition models.
\item Stochastic models: Markov chains and the Galton-Watson model.
\item Introduction to parameter estimation.
\item Bifurcution theory.
\item Enzyme kinetics.
\item Final numerical modeling projects and presentation.
\end{TopicsList}
