\begin{Text}
\emph{Advanced Calculus}, 5\textsuperscript{th} edition, by Kaplan,

\begin{TopicsList}
\item
  Vector operations, multiple integrals, line and surface integrals.
\item
  Vector operators: div, grad, and curl.
\item
  Jacobians and change of variables.
\item
  Green's Theorem, Stokes theorem.
\item
  Divergence Theorem.
\item
  Applications.
\end{TopicsList}

\begin{SuggestedSchedule}
\week{1}{Vector algebra; dot and cross products; lines and planes.}
\week{2}{Vector-valued functions; curves, arc length, curvature.}
\week{3}{Partial derivatives; the chain rule; the gradient.}
\week{4}{Double integrals; iterated integration.}
\week{5}{Triple integrals; cylindrical and spherical coordinates.}
\week{6}{Jacobians and change of variables. \emph{Midterm I.}}
\week{7}{Line integrals; work; conservative fields and potentials.}
\week{8}{Surfaces and surface integrals; flux.}
\week{9}{Divergence, gradient, and curl; vector identities.}
\week{10}{Green's Theorem.}
\week{11}{Stokes' Theorem. \emph{Midterm II.}}
\week{12}{The Divergence Theorem.}
\week{13}{Applications: fluid flow, gravitation, electromagnetism.}
\week{14}{Applications: potential theory; curvilinear coordinates; review.}
\finalsweek{Final examination.}
\end{SuggestedSchedule}
